_blockie_join() {
  local out=":" part d
  local -a dirs
  for part in "$@"; do
    IFS=: read -ra dirs <<< "$part"
    for d in "${dirs[@]}"; do
      [[ -n $d && $out != *":$d:"* ]] && out+="$d:"
    done
  done
  out=${out#:}
  printf '%s' "${out%:}"
}

_blockie_xlocks() {
  local dir=${BLOCKIE_XDIR:-/tmp} f n pid
  for f in "$dir"/.X*-lock; do
    n=${f##*/.X}
    n=${n%-lock}
    [[ $n =~ ^[0-9]+$ && -f $f && ! -L $f && -O $f ]] || continue
    pid=
    { read -r pid < "$f"; } 2> /dev/null
    [[ $pid =~ ^[1-9][0-9]*$ ]] && kill -0 "$pid" 2> /dev/null && continue
    rm -f -- "$f" "$dir/.X11-unix/X$n"
  done
}

_blockie_started() {
  local s
  local -a f
  { read -r s < "/proc/$1/stat"; } 2> /dev/null || return 1
  read -ra f <<< "${s##*) }"
  printf '%s' "${f[19]}"
}

_blockie_xwatch() {
  local dir=${BLOCKIE_XDIR:-/tmp} polls=${BLOCKIE_XWATCH_POLLS:-60} t0=${EPOCHREALTIME%[.,]*} me=$$ start i f s
  [[ $polls =~ ^[1-9][0-9]*$ ]] || polls=60
  start=$(_blockie_started "$me") || return 0
  for ((i = 0; i < polls; i++)); do
    sleep 0.5
    [[ $(_blockie_started "$me") == "$start" ]] || return 0
    for f in "$dir/.X11-unix/X0" "$dir/.X11-unix/X1"; do
      s=$(stat -c %Y -- "$f" 2> /dev/null) || continue
      (( s >= t0 )) && return 0
    done
  done
  [[ $(_blockie_started "$me") == "$start" ]] || return 0
  echo "blockie: westonpack display not up" >&2
  kill -TERM "$me" 2> /dev/null
}

BLOCKIE_ERROR_S=10

_blockie_portmsg() {
  env -i $BLOCKIE_PORTMSG_ENV /usr/libexec/blockie/blockie-portmsg "$@"
}

_blockie_words() {
  local t="${1,,}"
  if [[ $t == *"requires the latest portmaster"* || $t == *"requires the latest version of portmaster"* ||
    $t == *"requires a newer version of portmaster"* ]]; then
    printf '%s' "this port needs a newer blockie"
  else
    printf '%s' "$1"
  fi
}

_blockie_text() {
  local text="" title="" timeout=""
  if [ "${1#--}" = "$1" ]; then
    text="$1"
    shift
  fi
  while (( $# )); do
    case "$1" in
      --text) text="${2-}"; shift ;;
      --title) title="${2-}"; shift ;;
      --timeout) timeout="${2-}"; shift ;;
      --type | --ok-text | --cancel-text) shift ;;
      --*) ;;
      *) [[ -n $text ]] || text="$1" ;;
    esac
    shift
  done
  BLOCKIE_TEXT_ARGS=(--text "$(_blockie_words "$text")")
  [[ -n $title ]] && BLOCKIE_TEXT_ARGS+=(--title "$title")
  BLOCKIE_TEXT_TIMEOUT=""
  if [[ $timeout =~ ^[0-9]+$ ]]; then
    BLOCKIE_TEXT_TIMEOUT="$timeout"
    BLOCKIE_TEXT_ARGS+=(--timeout "$timeout")
  fi
}

_blockie_text_wait() {
  _blockie_text "$@"
  [[ -n $BLOCKIE_TEXT_TIMEOUT ]] || BLOCKIE_TEXT_ARGS+=(--timeout "$BLOCKIE_ERROR_S")
  _blockie_portmsg "${BLOCKIE_TEXT_ARGS[@]}" --wait
  local rc=$?
  (( rc == 3 )) && return 0
  return "$rc"
}

pm_message() {
  echo "$1"

  if [ "$1" = "Patching process already completed. Skipping." ]; then
    return
  fi

  _blockie_text "$@"
  [[ -n $BLOCKIE_TEXT_TIMEOUT ]] || BLOCKIE_TEXT_ARGS+=(--wait)
  _blockie_portmsg "${BLOCKIE_TEXT_ARGS[@]}" &
}

pm_show_error() {
  _blockie_text_wait "$@"
}

pm_message_close() {
  _blockie_portmsg --close || :
}

PortMasterDialogInit() {
  BLOCKIE_DIALOG_LAST=""
}

PortMasterDialogExit() {
  pm_message_close
}

PortMasterDialog() {
  BLOCKIE_DIALOG_RESULT="DONE"
  case "$1" in
    message)
      printf '%s\n' "$(_blockie_words "${2-}")" | _blockie_portmsg --progress || :
      ;;
    progress)
      local amount=0 total=100
      [[ ${3-} =~ ^[0-9]+$ ]] && amount="$3"
      [[ ${4-} =~ ^[1-9][0-9]*$ ]] && total="$4"
      if [[ -n ${2-} && $2 != "${BLOCKIE_DIALOG_LAST-}" ]]; then
        printf '%s\n' "$2" | _blockie_portmsg --progress || :
        BLOCKIE_DIALOG_LAST="$2"
      fi
      _blockie_portmsg --progress-set "$amount" "$total" || :
      ;;
    message_box)
      _blockie_text_wait "${@:2}"
      BLOCKIE_DIALOG_RESULT="TRUE"
      ;;
    check_runtime)
      if "$controlfolder/harbourmaster" --quiet --no-check runtime_check "${2-}"; then
        BLOCKIE_DIALOG_RESULT="OKAY"
      else
        BLOCKIE_DIALOG_RESULT="FAIL"
      fi
      ;;
  esac
}

PortMasterDialogResult() {
  PortMasterDialog "$@"
  echo "$BLOCKIE_DIALOG_RESULT"
}

PortMasterDialogMessageBox() {
  PortMasterDialog "message_box" "$@"
  echo "$BLOCKIE_DIALOG_RESULT"
}

PortMasterDialogCheckRuntime() {
  PortMasterDialog "check_runtime" "$@"
  echo "$BLOCKIE_DIALOG_RESULT"
}
